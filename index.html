async function fetchSalesData() { if (!currentStoreId) return; const startDate = document.getElementById('startDate').value; const endDate = document.getElementById('endDate').value; const loadingDiv = document.getElementById('loadingStatus'); loadingDiv.style.display = 'block'; try { const response = await fetch(`${SUPABASE_URL}/rest/v1/store_sales?store_id=eq.${currentStoreId}&order=last_update.desc&limit=2000&select=*`, { method: 'GET', headers: { 'apikey': SUPABASE_KEY, 'Authorization': `Bearer ${SUPABASE_KEY}`, 'Content-Type': 'application/json' } }); loadingDiv.style.display = 'none'; if (!response.ok) return; const data = await response.json(); let totalCash = 0; let sumCard = 0; let sumMobi = 0; let sumYesser = 0; let sumEdfaq = 0; let sumMultiGeneral = 0; let sumMultiSpecial = 0; let latestFormattedDateTime = "--/--/---- --:--:--"; if (data && data.length > 0) { const dailyRecords = {}; data.forEach(item => { if (!item.last_update) return; const itemDate = item.last_update.split('T')[0]; if (itemDate >= startDate && itemDate <= endDate) { if (!dailyRecords[itemDate]) { dailyRecords[itemDate] = item; } } }); if (data[0] && data[0].last_update) { const rawIsoString = data[0].last_update; const parsedDate = new Date(rawIsoString); if (!isNaN(parsedDate)) { const year = parsedDate.getFullYear(); const month = String(parsedDate.getMonth() + 1).padStart(2, '0'); const day = String(parsedDate.getDate()).padStart(2, '0'); const timeString = parsedDate.toTimeString().substring(0, 8); latestFormattedDateTime = `${year}/${month}/${day} - ${timeString}`; } } Object.values(dailyRecords).forEach(item => { const netCash = Number(item.cash_sales || 0) - Number(item.cash_returns || 0); const netCard = Number(item.card_sales || 0) - Number(item.card_returns || 0); const netMobi = Number(item.mobicash_sales || 0) - Number(item.mobicash_returns || 0); const netYesser = Number(item.yesserpay_sales || 0) - Number(item.yesserpay_returns || 0); const netEdfaq = Number(item.edfaqli_sales || 0) - Number(item.edfaqli_returns || 0); const netMultiGeneral = Number(item.multi_cash_sales || 0) - Number(item.multi_cash_returns || 0); // قراءة قيمة حساب 625 (كاش دفع متعدد) المحدثة من السحابة const netMultiSpecial = Number(item.multi_cash_special_sales || 0); totalCash += netCash; sumCard += netCard; sumMobi += netMobi; sumYesser += netYesser; sumEdfaq += netEdfaq; sumMultiGeneral += netMultiGeneral; sumMultiSpecial += netMultiSpecial; }); } const totalBank = sumCard + sumMobi + sumYesser + sumEdfaq; const totalMulti = sumMultiGeneral + sumMultiSpecial; // المجموع الكلي للدفع المتعدد (العام + 625) const grandTotal = totalCash + totalBank + totalMulti; currentBankBreakdown = { mobi: sumMobi, card: sumCard, yesser: sumYesser, edfaq: sumEdfaq }; // ربط القيم في نافذة تفاصيل الدفع المتعدد المنبثقة currentMultiBreakdown = { multiSpecial: sumMultiSpecial, multiGeneral: sumMultiGeneral }; document.getElementById('cashAmount').innerText = totalCash.toFixed(2) + " د.ل"; document.getElementById('bankAmount').innerText = totalBank.toFixed(2) + " د.ل"; document.getElementById('multiAmount').innerText = totalMulti.toFixed(2) + " د.ل"; document.getElementById('totalAmount').innerText = grandTotal.toFixed(2) + " د.ل"; document.getElementById('lastSyncTime').innerText = `آخر تحديث للبيانات: ${latestFormattedDateTime}`; if (document.getElementById('bankModal').style.display === 'block') toggleBankModal(true); if (document.getElementById('multiModal').style.display === 'block') toggleMultiModal(true); } catch (err) { loadingDiv.style.display = 'none'; console.error("خطأ في جلب المبيعات:", err); } }
